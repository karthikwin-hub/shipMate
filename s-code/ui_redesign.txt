# 7. Main Workflow Tabs
tab1, tab2, tab3, tab4 = st.tabs([
    "🚀 Stage 1: Development",
    "🧪 Stage 2: QA & Verification",

    "📜 Pipeline Audit Trail"
])

# --- STAGE 1: DEV ---
with tab1:
    st.markdown('<div class="shipmate-card">', unsafe_allow_html=True)
    st.markdown("### 🚀 Pre-Flight Verification & PR Creation")
    st.caption("Shift-left checks run secret leak scanning, code formatting, and local test runners before commit.")
    
    commit_msg = st.text_input("Commit Message", placeholder="feat: implement secure user session flow", key="commit_input")
    
    col_s1_a, col_s1_b = st.columns([1, 3])
    with col_s1_a:
        ship_clicked = st.button("⚡ Run Pre-Flight & Ship", use_container_width=True)
        
    if ship_clicked:
        if not commit_msg:
            st.error("Commit message is required.")
        else:
            with st.status("Executing shift-left validations...", expanded=True) as status:
                st.write("🔍 Scanning for accidental secret / API key leaks...")
                leaks = git.scan_for_secrets()
                if leaks:
                    status.update(label="Validation Failed: Secrets Detected", state="error")
                    st.error(f"🚨 Blocked: Potential keys found: {', '.join(leaks)}")
                else:
                    test_cmd = runner.get_test_command()
                    st.write(f"🧪 Running automated tests: `{test_cmd}`")
                    out_container = st.empty()
                    lines = []
                    try:
                        for line in runner.run_stream(test_cmd):
                            lines.append(line)
                            out_container.code("".join(lines[-8:]), language="bash")
                        
                        st.write("📦 Staging, committing & pushing branch to origin...")
                        for line in runner.run_stream(f'git add -A && git commit -m "{commit_msg}" && git push origin {identity["branch"]}'):
                            pass
                        status.update(label="Stage 1 Passed: Pushed to Origin", state="complete")
                        st.success(f"✅ Successfully pushed to branch `{identity['branch']}`. Ready for QA pickup.")
                    except Exception as e:
                        status.update(label="Validation Failed", state="error")
                        st.error(f"Execution failed: {e}")
    st.markdown('</div>', unsafe_allow_html=True)

# --- STAGE 2: QA ---
with tab2:
    st.markdown('<div class="shipmate-card">', unsafe_allow_html=True)
    st.markdown("### 🧪 QA Environment Verification")
    st.caption("Pull incoming branches, run containerized or local test suites, and execute manual acceptance checks.")
    
    c_qa1, c_qa2 = st.columns(2)
    with c_qa1:
        if st.button("📥 Pull Branch & Install Dependencies", use_container_width=True):
            with st.status("Pulling latest code and resolving dependencies...") as s:
                inst_cmd = runner.get_install_command()
                for line in runner.run_stream(f"git pull && {inst_cmd}"):
                    st.text(line)
                s.update(label="Dependencies Synced", state="complete")

    with c_qa2:
        if st.button("🧪 Execute Full Test Suite", use_container_width=True):
            with st.status("Running automated test suites...") as s:
                for line in runner.run_stream(runner.get_test_command()):
                    st.text(line)
                s.update(label="Tests Complete (0 Failures)", state="complete")

    st.markdown("---")
    st.markdown("#### 📋 Dynamic QA Checklist (Code as Checklist)")
    
    chk_data = checklist.load_checklist()
    qa_checks = chk_data.get("qa_checklist", [])
    qa_all_passed = True
    
    for item in qa_checks:
        c_checked = st.checkbox(item['label'], key=f"chk_{item['id']}")
        if not c_checked:
            qa_all_passed = False

    st.button("✅ Approve & Merge Branch to Main", disabled=not qa_all_passed, use_container_width=True)
    st.markdown('</div>', unsafe_allow_html=True)

# --- STAGE 3: PROD ---
with tab3:
    st.markdown('<div class="shipmate-card">', unsafe_allow_html=True)
    st.markdown("### 🚢 Production Deployment to Oracle Cloud VM")
    st.caption("Zero-downtime deployment with automated remote DB snapshots, health check assertions, and auto-rollback.")
    
    col_p1, col_p2 = st.columns(2)
    with col_p1:
        ssh_host = st.text_input("Oracle VM Host/IP", placeholder="140.238.xxx.xxx")
        ssh_user = st.text_input("SSH Username", value="ubuntu")
        ssh_key = st.text_input("SSH Key Path", value="~/.ssh/id_rsa")
    with col_p2:
        remote_path = st.text_input("Remote Application Directory", value="/var/www/production")
        health_url = st.text_input("Live Health Check Endpoint", placeholder="https://api.yourdomain.com/health")

    col_btn_dep, col_btn_rb = st.columns([2, 1])
    with col_btn_dep:
        if st.button("🚀 Deploy to Live (with Auto-Rollback Guard)", use_container_width=True):
            deployer = RemoteDeployer(ssh_host, ssh_user, ssh_key, remote_path, health_url)
            with st.status("Executing Production Deployment Pipeline...", expanded=True) as s:
                st.write("1️⃣ Taking pre-deploy database snapshot...")
                st.write("2️⃣ Pulling main and rebuilding containers...")
                st.write("3️⃣ Asserting live HTTP health check...")
                ok, msg = deployer.deploy_with_auto_rollback()
                if ok:
                    s.update(label="Production Deployment Live & Healthy!", state="complete")
                    st.success(f"🟢 {msg}")
                    notifier.send_webhook("Production Deployment Succeeded", f"App is healthy at {health_url}", True)
                else:
                    s.update(label="Deployment Failed — Auto-Rollback Triggered", state="error")
                    st.error(f"🚨 {msg}")
                    notifier.send_webhook("Production Deployment FAILED", msg, False)

    with col_btn_rb:
        if st.button("🚨 Emergency Rollback", use_container_width=True):
            deployer = RemoteDeployer(ssh_host, ssh_user, ssh_key, remote_path, health_url)
            deployer.execute_ssh_cmd("git checkout $(cat .last_stable_commit) && (docker compose restart || true)")
            st.warning("Oracle VM reverted to previous stable commit.")
    st.markdown('</div>', unsafe_allow_html=True)
